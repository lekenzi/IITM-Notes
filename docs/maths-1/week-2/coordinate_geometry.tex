\section*{Distance formula}
\[
  d = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}
\]
\begin{center}
\begin{tikzpicture}
  \draw[->] (-0.5,0) -- (4,0) node[right] {$x$};
  \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};
  \fill (1,1) circle (2pt) node[below left] {$(x_1,y_1)$};
  \fill (3,2.5) circle (2pt) node[above right] {$(x_2,y_2)$};
  \draw[thick] (1,1) -- (3,2.5) node[midway, above left] {$d$};
\end{tikzpicture}
\end{center}
